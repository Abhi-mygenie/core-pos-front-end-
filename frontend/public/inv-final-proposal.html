toBreakdownDisplay() — to be implemented at Gate 4 GO
// Converts a base-unit value to "8 bottle 319 ml" style text
// Same format as backend's display_qty_text field
function toBreakdownDisplay(baseValue, convFactor, displayUnit, smallUnit) {
  if (!convFactor || convFactor <= 0)
    return `${baseValue} ${smallUnit}`;
  const abs = Math.abs(baseValue);
  const whole = Math.floor(abs / convFactor);
  const remainder = Math.round(abs % convFactor);
  const sign = baseValue < 0 ? '-' : '';
  if (remainder === 0) return `${sign}${whole} ${displayUnit}`;
  return `${sign}${whole} ${displayUnit} ${remainder} ${smallUnit}`;
}

// Examples with real yabyum data:
// toBreakdownDisplay(422500, 650, 'bottle', 'ml') → "650 bottle"
// toBreakdownDisplay(416980, 650, 'bottle', 'ml') → "641 bottle 330 ml"
// toBreakdownDisplay(1600000, 1598.25, 'tin', 'gm') → "1001 tin 152 gm"
// toBreakdownDisplay(-10000, 500, 'pkt', 'gm') → "-20 pkt"
// toBreakdownDisplay(2290000, 500, 'pkt', 'gm') → "4580 pkt" (live input conversion)
